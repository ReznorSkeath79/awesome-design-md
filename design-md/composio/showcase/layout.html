reconcile.ts
import { Northwind } from "@northwind/sdk";

const nw = new Northwind({ region: "luzon" });

const report = await nw.agents.reconcile({
  lines: "all",
  week: 41,
});

// 14 lines scanned, flagged where yield drifted
console.log(report.flagged);
